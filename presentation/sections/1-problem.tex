\begin{frame}{Treść problemu}
  Cel: wybór lidera przy następujących założeniach:
      \pause
    \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
      \begin{enumerate}[<+->]
        \item Topologia: skierowany $n$-cykl.
        \item Obliczenia są synchroniczne. W każdej rundzie węzeł:
          \begin{enumerate}[<+->]
            \item odbiera wiadomość,
            \item zmienia swój stan,
            \item może wysłać jedną wiadomość odczytaną w następnej rundzie
          \end{enumerate}
        \item Wiemy jedynie, że $N \leq n \leq 2N-1$, gdzie $N$ jest znane
          każdemu węzłowi.
        \item Węzły mają dostęp do niezależnych ciągów losowych liczb
          rzeczywistych o rozkładzie jednostajnym na $[0, 1]$.
      \end{enumerate}
    \end{column}

          \draw[dlink] (v\i) to[bend left=20] (v\j);
        }
